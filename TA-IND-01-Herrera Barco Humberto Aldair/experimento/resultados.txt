=====================================================================
2. Índice de pruebas, distribución de sharts
=====================================================================
Comandos:
Set-Content indice.json '{"settings":{"number_of_shards":3,"number_of_replicas":1}}' -Encoding ascii
curl.exe -X PUT "localhost:9200/pruebas" -H "Content-Type: application/json" -d "@indice.json"
1..6 | ForEach-Object { Set-Content doc.json ('{"mensaje":"documento ' + $_ + '"}') -Encoding ascii; curl.exe -s -X POST "localhost:9200/pruebas/_doc" -H "Content-Type: application/json" -d "@doc.json" }
whoami; hostname; date; curl.exe "localhost:9200/_cat/shards/pruebas?v"
curl.exe -X POST "localhost:9200/pruebas/_refresh"
whoami; hostname; date; curl.exe "localhost:9200/_cat/shards/pruebas?v"
